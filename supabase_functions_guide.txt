-- com a plataforma de pagamentos Asaas.

-- ANTES DE COMEÇAR:
-- 1. Configure as variáveis de ambiente no seu projeto Supabase:
--    - SUPABASE_URL: A URL do seu projeto.
--    - SUPABASE_SERVICE_ROLE_KEY: Sua chave de serviço (segura).
--    - ASAAS_API_KEY: Sua chave de API do Asaas (produção ou sandbox).
--    - ASAAS_WEBHOOK_SECRET: Uma chave secreta que você cria para validar webhooks.

-- 2. Instale o Supabase CLI: `npm install supabase --save-dev`
-- 3. Faça login: `supabase login`
-- 4. Vincule seu projeto: `supabase link --project-ref <SEU_PROJECT_ID>`

-- FUNÇÃO 2: `process-payment` (Para Checkout da Loja) - [NOVO]
-- Finalidade: Processa um pedido da loja, criando a cobrança no Asaas.
-- Comando: `supabase functions deploy process-payment`
-- ==================================================================
-- Arquivo: `supabase/functions/process-payment/index.ts`
/*
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient, SupabaseClient } from "https://esm.sh/@supabase/supabase-js@2";

const ASAAS_API_KEY = Deno.env.get("ASAAS_API_KEY");
const ASAAS_URL = "https://api.asaas.com/v3";

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" } });
  }

  try {
    const supabase = createClient(Deno.env.get("SUPABASE_URL") ?? "", Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "", {
      global: { headers: { Authorization: req.headers.get("Authorization")! } },
    });
    
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) throw new Error("Usuário não autenticado.");

    const { items, total_price, payment_method, shipping_address, payment_details } = await req.json();

    // 1. Get user profile for Asaas customer ID
    const { data: profile, error: profileError } = await supabase
      .from('user_profiles')
      .select('asaas_customer_id, name, email, phone_number')
      .eq('id', user.id)
      .single();

    if (profileError) throw profileError;

    let customerId = profile.asaas_customer_id;
    // Create Asaas customer if not exists
    if (!customerId) {
        // ... Lógica para criar cliente no Asaas e atualizar profile.asaas_customer_id ...
    }
    
    // 2. Create Order in DB (status: pending_payment)
    const { data: order, error: orderError } = await supabase
      .from('orders')
      .insert({ user_id: user.id, items, total_price, payment_method, shipping_address, status: 'pending_payment' })
      .select()
      .single();
      
    if (orderError) throw orderError;
    
    // 3. Create Asaas Charge
    const chargePayload = {
      customer: customerId,
      billingType: payment_method,
      value: total_price,
      dueDate: new Date(new Date().setDate(new Date().getDate() + 3)).toISOString().split('T')[0],
      description: `Pedido #${order.id.substring(0,8)} - Zé Entregas`,
      externalReference: order.id,
      // ... Add credit card details if payment_method is CREDIT_CARD
    };

    const response = await fetch(`${ASAAS_URL}/payments`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'access_token': ASAAS_API_KEY },
      body: JSON.stringify(chargePayload),
    });

    const asaasData = await response.json();
    if (!response.ok) throw new Error(asaasData.errors?.[0]?.description || 'Falha ao criar cobrança.');

    // 4. Update order with Asaas data
    const { data: updatedOrder, error: updateError } = await supabase
      .from('orders')
      .update({ 
          asaas_id: asaasData.id,
          asaas_invoice_url: asaasData.invoiceUrl,
          asaas_pix_qrcode: asaasData.pixQrCode,
          asaas_pix_copy_paste: asaasData.pixCopyPaste,
          asaas_bank_slip_url: asaasData.bankSlipUrl,
          due_date: asaasData.dueDate
      })
      .eq('id', order.id)
      .select()
      .single();
      
    if (updateError) throw updateError;

    return new Response(JSON.stringify(updatedOrder), { status: 200, headers: { "Content-Type": "application/json" } });

  } catch (e) {
    return new Response(JSON.stringify({ error: e.message }), { status: 400, headers: { "Content-Type": "application/json" } });
  }
});
