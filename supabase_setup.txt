-- Função para verificar se o usuário é admin
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN (SELECT role FROM public.user_profiles WHERE id = auth.uid()) = 'admin';
END;
$$;

-- Função para verificar se o usuário é o dono do perfil
CREATE OR REPLACE FUNCTION public.is_owner(p_user_id uuid)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN auth.uid() = p_user_id;
END;
$$;

-- Trigger para criar perfil de usuário e carteiras após novo registro no auth.users
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    user_role public.user_role_enum;
    user_phone TEXT;
    user_cpf TEXT;
    user_city TEXT;
    user_name TEXT;
    user_avatar_url TEXT;
BEGIN
    user_role := COALESCE(NEW.raw_user_meta_data->>'role', 'delivery_person')::public.user_role_enum;
    user_phone := NEW.raw_user_meta_data->>'phone';
    user_cpf := NEW.raw_user_meta_data->>'cpf';
    user_city := NEW.raw_user_meta_data->>'city';
    user_name := NEW.raw_user_meta_data->>'name';
    user_avatar_url := NEW.raw_user_meta_data->>'avatar_url';

    INSERT INTO public.user_profiles (id, name, email, phone_number, cpf, city, avatar_url, role)

CREATE OR REPLACE FUNCTION public.get_active_drivers_nearby(p_lat numeric, p_lng numeric, p_radius_km integer)
RETURNS TABLE (id uuid, current_lat numeric, current_lng numeric, distance_m integer, vehicle_type public.vehicle_type_enum, average_rating numeric, partner_level text, is_super_store boolean)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT
        up.id,
        (up.user_metadata->>'current_lat')::numeric AS current_lat,
        (up.user_metadata->>'current_lng')::numeric AS current_lng,
        (ST_DistanceSphere(
            ST_SetSRID(ST_MakePoint(p_lng, p_lat), 4326),
            ST_SetSRID(ST_MakePoint((up.user_metadata->>'current_lng')::numeric, (up.user_metadata->>'current_lat')::numeric), 4326)
        ))::integer AS distance_m,
        up.vehicle_type,
        up.average_rating,
        up.partner_level,
        up.is_super_store
    FROM
        public.user_profiles up
    WHERE
        (up.role = 'delivery_partner' OR up.role = 'delivery_person')
        AND up.status = 'active'
        AND up.is_available = true
        AND up.user_metadata->>'current_lat' IS NOT NULL
        AND up.user_metadata->>'current_lng' IS NOT NULL
        AND ST_DistanceSphere(
            ST_SetSRID(ST_MakePoint(p_lng, p_lat), 4326),
            ST_SetSRID(ST_MakePoint((up.user_metadata->>'current_lng')::numeric, (up.user_metadata->>'current_lat')::numeric), 4326)
        ) <= p_radius_km * 1000
    ORDER BY
        distance_m;
END;
$$;

CREATE OR REPLACE FUNCTION public.create_partner_request(
    p_pickup_address text,
    p_delivery_address text,
    p_distance_km numeric,
    p_final_cost_override numeric DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    store_profile public.user_profiles;
    payout_settings public.payout_settings;
    calculated_partner_net numeric;
    calculated_fee_fixed numeric;
    calculated_fee_percent_value numeric;
    final_total_charged_store numeric;
    new_request_id uuid;
    store_wallet public.store_wallets;
BEGIN
    SELECT * INTO store_profile FROM public.user_profiles WHERE id = auth.uid();
    IF store_profile.role IS DISTINCT FROM 'store_partner' THEN
        RAISE EXCEPTION 'Apenas lojistas podem criar solicitações de parceiro.';
    END IF;

    SELECT * INTO payout_settings FROM public.payout_settings LIMIT 1;
    IF payout_settings IS NULL THEN
        RAISE EXCEPTION 'Configurações de repasse não encontradas.';
    END IF;

    calculated_partner_net := payout_settings.base_delivery_value;
    IF p_distance_km > payout_settings.base_delivery_km THEN
        calculated_partner_net := calculated_partner_net + (p_distance_km - payout_settings.base_delivery_km) * payout_settings.extra_km_value;
    END IF;

    IF p_delivery_address LIKE '% -> %' AND payout_settings.additional_stop_fee IS NOT NULL THEN
        calculated_partner_net := calculated_partner_net + (LENGTH(p_delivery_address) - LENGTH(REPLACE(p_delivery_address, ' -> ', ''))) / LENGTH(' -> ') * payout_settings.additional_stop_fee;
    END IF;

    calculated_fee_fixed := payout_settings.global_tax_fixed;
    calculated_fee_percent_value := calculated_partner_net * payout_settings.global_tax_percent;

    final_total_charged_store := calculated_partner_net + calculated_fee_fixed + calculated_fee_percent_value;

    IF p_final_cost_override IS NOT NULL THEN
        final_total_charged_store := p_final_cost_override;
        calculated_partner_net := final_total_charged_store / (1 + payout_settings.global_tax_percent) - payout_settings.global_tax_fixed;
        calculated_fee_fixed := payout_settings.global_tax_fixed;
        calculated_fee_percent_value := final_total_charged_store - calculated_partner_net - calculated_fee_fixed;
    END IF;

    SELECT * INTO store_wallet FROM public.store_wallets WHERE store_id = auth.uid();
    IF store_wallet.balance_decimal < final_total_charged_store THEN
        RAISE EXCEPTION 'Saldo insuficiente na carteira para criar esta solicitação. Saldo atual: %, Necessário: %', store_wallet.balance_decimal, final_total_charged_store;
    END IF;

    UPDATE public.store_wallets
    SET
        balance = balance - final_total_charged_store,
        balance_decimal = balance_decimal - final_total_charged_store
    WHERE store_id = auth.uid();

    INSERT INTO public.store_wallet_transactions (store_id, amount, type, status, description)
    VALUES (auth.uid(), -final_total_charged_store, 'partner_request_fee', 'confirmed', 'Débito para solicitação de entregador');

    INSERT INTO public.partner_requests (
        store_id, pickup_address, delivery_address, distance_km,
        total_charged_store, net_value_partner, fee_fixed, fee_percent_value
    )
    VALUES (
        auth.uid(), p_pickup_address, p_delivery_address, p_distance_km,
        final_total_charged_store, calculated_partner_net, calculated_fee_fixed, calculated_fee_percent_value
    )
    RETURNING id INTO new_request_id;

    RETURN new_request_id;
END;
$$;

CREATE OR REPLACE FUNCTION public.get_store_requests()
RETURNS TABLE (
    id uuid,
    store_id uuid,
    partner_id uuid,
    pickup_address text,
    delivery_address text,
    distance_km numeric,
    total_charged_store numeric,
    net_value_partner numeric,
    fee_fixed numeric,
    fee_percent_value numeric,
    status public.partner_request_status_enum,
    created_at timestamp with time zone,
    updated_at timestamp with time zone,
    failure_reason text,
    rated_by_store boolean,
    rated_by_partner boolean,
    store_decision_status public.request_decision_status_enum,
    partner jsonb,
    store jsonb

LANGUAGE plpgsql
SECURITY DEFINER

BEGIN

        pr.id,

        pr.delivery_address,
        pr.distance_km,
        pr.total_charged_store,
        pr.net_value_partner,
        pr.fee_fixed,
        pr.fee_percent_value,

        pr.created_at,
        pr.updated_at,
        pr.failure_reason,
        pr.rated_by_store,

CREATE OR REPLACE FUNCTION public.partner_confirm_delivery(p_request_id uuid)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    current_status public.partner_request_status_enum;
    request_details public.partner_requests;
    partner_wallet_record public.partner_wallets;
    store_wallet_record public.store_wallets;
BEGIN
    SELECT * INTO request_details FROM public.partner_requests WHERE id = p_request_id;

    IF request_details IS NULL THEN
        RAISE EXCEPTION 'Solicitação não encontrada.';
    END IF;

    IF request_details.partner_id IS DISTINCT FROM auth.uid() THEN
        RAISE EXCEPTION 'Você não é o parceiro desta solicitação.';
    END IF;

    IF request_details.status <> 'IN_TRANSIT' THEN
        RAISE EXCEPTION 'A solicitação não está no status correto para entrega (status atual: %).', request_details.status;
    END IF;

    UPDATE public.partner_requests
    SET status = 'COMPLETED'
    WHERE id = p_request_id;

    UPDATE public.partner_wallets
    SET balance = balance + request_details.net_value_partner
    WHERE partner_id = auth.uid()
    RETURNING * INTO partner_wallet_record;

    IF NOT FOUND THEN
        INSERT INTO public.partner_wallets (partner_id, balance) VALUES (auth.uid(), request_details.net_value_partner);
    END IF;

    INSERT INTO public.partner_payments (partner_id, amount, is_emergency, status)
    VALUES (auth.uid(), request_details.net_value_partner, false, 'EARNED');

    UPDATE public.user_profiles
    SET completed_deliveries = COALESCE(completed_deliveries, 0) + 1
    WHERE id = auth.uid();

END;
$$;

CREATE OR REPLACE FUNCTION public.store_decide_failed_delivery(p_request_id uuid, p_decision public.request_decision_status_enum)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    request_details public.partner_requests;
BEGIN
    SELECT * INTO request_details FROM public.partner_requests WHERE id = p_request_id;

    IF request_details IS NULL THEN
        RAISE EXCEPTION 'Solicitação não encontrada.';
    END IF;

    IF request_details.store_id IS DISTINCT FROM auth.uid() THEN
        RAISE EXCEPTION 'Você não é o lojista desta solicitação.';
    END IF;

    IF request_details.status IS DISTINCT FROM 'AWAITING_STORE_DECISION' THEN
        RAISE EXCEPTION 'A solicitação não está aguardando decisão do lojista (status atual: %).', request_details.status;
    END IF;

    IF p_decision = 'RETURN' THEN
        UPDATE public.partner_requests
        SET status = 'RETURNING', store_decision_status = 'RETURN'
        WHERE id = p_request_id;
    ELSIF p_decision = 'DISCARD' THEN
        UPDATE public.partner_requests
        SET status = 'COMPLETED', store_decision_status = 'DISCARD'
        WHERE id = p_request_id;

        UPDATE public.partner_wallets
        SET balance = balance + request_details.net_value_partner
        WHERE partner_id = request_details.partner_id;

        INSERT INTO public.partner_payments (partner_id, amount, is_emergency, status)
        VALUES (request_details.partner_id, request_details.net_value_partner, false, 'EARNED_DISCARDED');
    ELSE
        RAISE EXCEPTION 'Decisão inválida: %', p_decision;
    END IF;

END;
$$;

CREATE OR REPLACE FUNCTION public.get_partner_financial_summary()
RETURNS TABLE (
    total_earnings numeric,
    total_withdrawals numeric,
    current_balance numeric,
    max_emergency_value numeric,
    can_request_emergency boolean,
    settings jsonb
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    p_id uuid := auth.uid();
    p_settings public.payout_settings;
    p_wallet public.partner_wallets;
    last_emergency_payout timestamp with time zone;
    total_payments numeric;
    total_earnings_sum numeric;
BEGIN
    SELECT * INTO p_settings FROM public.payout_settings LIMIT 1;
    IF p_settings IS NULL THEN
        RAISE EXCEPTION 'Configurações de repasse não encontradas.';
    END IF;

    SELECT * INTO p_wallet FROM public.partner_wallets WHERE partner_id = p_id;
    IF p_wallet IS NULL THEN
        current_balance := 0;
    ELSE
        current_balance := p_wallet.balance;
    END IF;

    SELECT COALESCE(SUM(net_value_partner), 0)
    INTO total_earnings_sum
    FROM public.partner_requests
    WHERE partner_id = p_id AND status = 'COMPLETED';

    SELECT COALESCE(SUM(amount), 0)
    INTO total_payments
    FROM public.partner_payments
    WHERE partner_id = p_id AND status = 'DONE';

    total_earnings := total_earnings_sum;
    total_withdrawals := total_payments;

    max_emergency_value := 0;
    can_request_emergency := false;

    IF p_settings.emergency_enabled THEN
        max_emergency_value := current_balance * (p_settings.emergency_percentage / 100.0);

        SELECT MAX(created_at)
        INTO last_emergency_payout
        FROM public.partner_payments
        WHERE partner_id = p_id AND is_emergency = true AND status = 'DONE';

        IF last_emergency_payout IS NULL OR (NOW() - last_emergency_payout) >= (p_settings.emergency_cooldown_hours || ' hours')::interval THEN
            can_request_emergency := true;
        END IF;
    END IF;

    RETURN QUERY SELECT
        total_earnings,
        total_withdrawals,
        current_balance,
        max_emergency_value,
        can_request_emergency,
        to_jsonb(p_settings) AS settings_jsonb;
END;
$$;

CREATE OR REPLACE FUNCTION public.submit_rating(
    p_request_id uuid,
    p_evaluator_id uuid,
    p_rating integer,
    p_comment text,
    p_direction public.rating_direction_enum
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    req_store_id uuid;
    req_partner_id uuid;
    evaluated_user_id uuid;
BEGIN
    SELECT store_id, partner_id INTO req_store_id, req_partner_id FROM public.partner_requests WHERE id = p_request_id;

    IF p_direction = 'STORE_TO_PARTNER' THEN
        IF p_evaluator_id IS DISTINCT FROM req_store_id THEN
            RAISE EXCEPTION 'Apenas o lojista pode avaliar o parceiro nesta direção.';
        END IF;
        evaluated_user_id := req_partner_id;
        UPDATE public.partner_requests SET rated_by_store = true WHERE id = p_request_id;
    ELSIF p_direction = 'PARTNER_TO_STORE' THEN
        IF p_evaluator_id IS DISTINCT FROM req_partner_id THEN
            RAISE EXCEPTION 'Apenas o parceiro pode avaliar o lojista nesta direção.';
        END IF;
        evaluated_user_id := req_store_id;
        UPDATE public.partner_requests SET rated_by_partner = true WHERE id = p_request_id;
    ELSE
        RAISE EXCEPTION 'Direção de avaliação inválida.';
    END IF;

    INSERT INTO public.partner_ratings (request_id, evaluator_id, evaluated_id, rating, comment, direction)
    VALUES (p_request_id, p_evaluator_id, evaluated_user_id, p_rating, p_comment, p_direction);

    UPDATE public.user_profiles
    SET average_rating = (
        SELECT AVG(rating) FROM public.partner_ratings WHERE evaluated_id = evaluated_user_id
    )
    WHERE id = evaluated_user_id;
END;
$$;

CREATE OR REPLACE FUNCTION public.get_store_reports()
RETURNS TABLE (
    total_requests bigint,
    total_value numeric,
    peak_hours jsonb,
    driver_performance jsonb
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    store_id uuid := auth.uid();
    today_start timestamp with time zone := date_trunc('day', now());
    last_30_days_start timestamp with time zone := now() - INTERVAL '30 days';
BEGIN
    RETURN QUERY
    SELECT
        COUNT(pr.id) AS total_requests,
        COALESCE(SUM(pr.total_charged_store), 0) AS total_value,
        COALESCE(jsonb_agg(jsonb_build_object('hour', h.hour, 'count', h.count) ORDER BY h.hour), '[]'::jsonb) AS peak_hours,
        COALESCE(jsonb_agg(jsonb_build_object('partner_id', dp.partner_id, 'partner_name', up.name, 'count', dp.delivery_count) ORDER BY dp.delivery_count DESC), '[]'::jsonb) AS driver_performance
    FROM
        public.partner_requests pr
    LEFT JOIN (
        SELECT
            EXTRACT(HOUR FROM created_at) AS hour,
            COUNT(id) AS count
        FROM
            public.partner_requests
        WHERE
            store_id = store_id AND created_at >= today_start
        GROUP BY
            EXTRACT(HOUR FROM created_at)
    ) h ON true
    LEFT JOIN (
        SELECT
            partner_id,
            COUNT(id) AS delivery_count
        FROM
            public.partner_requests
        WHERE
            store_id = store_id AND created_at >= last_30_days_start AND status = 'COMPLETED'
        GROUP BY
            partner_id
    ) dp ON true
    LEFT JOIN
        public.user_profiles up ON dp.partner_id = up.id
    WHERE
        pr.store_id = store_id AND pr.created_at >= last_30_days_start;
END;
$$;

CREATE OR REPLACE FUNCTION public.admin_adjust_balance(p_user_id uuid, p_amount numeric, p_reason text)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    user_role public.user_role_enum;
BEGIN
    IF NOT public.is_admin() THEN
        RAISE EXCEPTION 'Acesso negado: Somente administradores podem ajustar saldos.';
    END IF;

    SELECT role INTO user_role FROM public.user_profiles WHERE id = p_user_id;

    IF user_role = 'store_partner' THEN
        UPDATE public.store_wallets
        SET
            balance = balance + p_amount,
            balance_decimal = balance_decimal + p_amount
        WHERE store_id = p_user_id;

        INSERT INTO public.store_wallet_transactions (store_id, amount, type, status, description)
        VALUES (p_user_id, p_amount, 'adjustment', 'confirmed', 'Ajuste Admin: ' || p_reason);

    ELSIF user_role = 'delivery_partner' OR user_role = 'delivery_person' THEN
        UPDATE public.partner_wallets
        SET balance = balance + p_amount
        WHERE partner_id = p_user_id;

        INSERT INTO public.partner_payments (partner_id, amount, is_emergency, status)
        VALUES (p_user_id, p_amount, false, 'ADJUSTMENT_ADMIN');
    ELSE
        RAISE EXCEPTION 'Tipo de usuário não suportado para ajuste de saldo.';
    END IF;
END;
$$;

CREATE OR REPLACE FUNCTION public.associate_partner_to_store(p_partner_id uuid, p_fee numeric DEFAULT 0.0)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    store_profile public.user_profiles;
    partner_profile public.user_profiles;
    store_wallet public.store_wallets;
BEGIN
    SELECT * INTO store_profile FROM public.user_profiles WHERE id = auth.uid();
    IF store_profile.role IS DISTINCT FROM 'store_partner' THEN
        RAISE EXCEPTION 'Apenas lojistas podem associar entregadores.';
    END IF;

    SELECT * INTO partner_profile FROM public.user_profiles WHERE id = p_partner_id;
    IF partner_profile.role IS DISTINCT FROM 'delivery_partner' AND partner_profile.role IS DISTINCT FROM 'delivery_person' THEN
        RAISE EXCEPTION 'O ID fornecido não pertence a um entregador parceiro.';
    END IF;

    IF EXISTS (SELECT 1 FROM public.store_delivery_partners WHERE store_id = auth.uid() AND partner_id = p_partner_id) THEN
        RAISE EXCEPTION 'Este entregador já está associado à sua loja.';
    END IF;

    IF p_fee > 0 THEN
        SELECT * INTO store_wallet FROM public.store_wallets WHERE store_id = auth.uid();
        IF store_wallet.balance_decimal < p_fee THEN
            RAISE EXCEPTION 'Saldo insuficiente na carteira para pagar a taxa de associação. Saldo atual: %, Necessário: %', store_wallet.balance_decimal, p_fee;
        END IF;

        UPDATE public.store_wallets
        SET
            balance = balance - p_fee,
            balance_decimal = balance_decimal - p_fee
        WHERE store_id = auth.uid();

        INSERT INTO public.store_wallet_transactions (store_id, amount, type, status, description)
        VALUES (auth.uid(), -p_fee, 'association_fee', 'confirmed', 'Débito para taxa de associação de entregador');
    END IF;

    INSERT INTO public.store_delivery_partners (store_id, partner_id, fee_override)
    VALUES (auth.uid(), p_partner_id, p_fee);
END;
$$;

CREATE OR REPLACE FUNCTION public.redeem_referral_code(p_code text)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE

        SET referral_reward_expires_at = COALESCE(referral_reward_expires_at, now()) + reward_duration
        WHERE id = referrer_id;

    VALUES (referrer_id, auth.uid(), 'REWARDED');

    IF referrer_role = 'store_partner' THEN
    ELSIF referrer_role = 'delivery_partner' OR referrer_role = 'delivery_person' THEN
        UPDATE public.user_profiles
        SET referral_reward_expires_at = COALESCE(referral_reward_expires_at, now()) + reward_duration
        WHERE id = auth.uid();
    END IF;

END;
$$;


CREATE OR REPLACE FUNCTION public.resolve_login_email(identifier text)
 RETURNS text
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $$
BEGIN
    IF identifier LIKE '%@%' THEN
        RETURN identifier;
    END IF;

    IF LENGTH(REPLACE(REPLACE(REPLACE(identifier, '(', ''), ')', ''), '-', '')) >= 8 AND LENGTH(REPLACE(REPLACE(REPLACE(identifier, '(', ''), ')', ''), '-', '')) <= 11 THEN
        RETURN (SELECT email FROM public.user_profiles WHERE phone_number = identifier LIMIT 1);
    END IF;

    IF LENGTH(REPLACE(REPLACE(identifier, '.', ''), '-', '')) = 11 THEN
        RETURN (SELECT email FROM public.user_profiles WHERE cpf = identifier LIMIT 1);
    END IF;

    RETURN NULL;
END;
$$;

CREATE OR REPLACE FUNCTION public.subscribe_to_super_store()
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    fee numeric;
    current_balance numeric;
BEGIN
    -- Get Fee from settings
    SELECT super_store_monthly_fee INTO fee FROM public.payout_settings LIMIT 1;

    IF fee IS NULL THEN
        RAISE EXCEPTION 'Taxa de assinatura não configurada.';
    END IF;

    -- Get Balance
    SELECT balance_decimal INTO current_balance FROM public.store_wallets WHERE store_id = auth.uid();

    IF current_balance IS NULL OR current_balance < fee THEN
        RAISE EXCEPTION 'Saldo insuficiente. Realize uma recarga.';
    END IF;

    -- Deduct
    UPDATE public.store_wallets
    SET balance = balance - fee, balance_decimal = balance_decimal - fee
    WHERE store_id = auth.uid();

    -- Log Transaction
    INSERT INTO public.store_wallet_transactions (store_id, amount, type, status, description)
    VALUES (auth.uid(), -fee, 'subscription', 'confirmed', 'Assinatura Super Lojista (Mensal)');

    -- Update Profile
    UPDATE public.user_profiles SET is_super_store = true WHERE id = auth.uid();
END;
$$;

-- Tabela de Cartões Zebank
CREATE TABLE IF NOT EXISTS public.zebank_cards (
    id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id uuid NOT NULL REFERENCES public.user_profiles(id) ON DELETE CASCADE,
    card_last_four text NOT NULL,
    card_holder text NOT NULL,
    expiration_date text NOT NULL,
    cvv text, -- Armazenado apenas para simulação visual neste MVP, em prod real seria token
    type text DEFAULT 'VIRTUAL' CHECK (type IN ('VIRTUAL', 'PHYSICAL')),
    status text DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE', 'BLOCKED')),
    balance_limit numeric DEFAULT 0.0, -- Limite de gasto do cartão (opcional, vinculado à carteira)
    created_at timestamp with time zone DEFAULT now() NOT NULL
);
ALTER TABLE public.zebank_cards ENABLE ROW LEVEL SECURITY;

-- Adicionar colunas se faltarem em zebank_cards (caso a tabela já existisse de forma simples)
DO $$ BEGIN
    IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='zebank_cards' AND column_name='cvv') THEN
        ALTER TABLE public.zebank_cards ADD COLUMN cvv text;
    END IF;
END $$;
