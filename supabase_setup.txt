-- Helper: Verificar se o usuário autenticado é Admin
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS boolean AS $$
BEGIN
  RETURN EXISTS (
    SELECT 1 FROM public.user_profiles 
    WHERE id = auth.uid() AND role = 'admin'::public.user_role
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5.1 Resolver Email no Login (User, CPF ou Phone)
CREATE OR REPLACE FUNCTION public.resolve_login_email(identifier text)
RETURNS text AS $$
DECLARE
    found_email text;
BEGIN
    SELECT email INTO found_email FROM public.user_profiles 
    WHERE email = identifier 
       OR phone_number = identifier 
       OR cpf = identifier
    LIMIT 1;

    RETURN found_email;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5.2 Upgrade de Usuário para Parceiro Entregador
CREATE OR REPLACE FUNCTION public.become_delivery_partner()
RETURNS void AS $$
DECLARE
    v_user_id uuid := auth.uid();
BEGIN
    -- Verifica se já é parceiro
    IF EXISTS (SELECT 1 FROM public.user_profiles WHERE id = v_user_id AND role = 'delivery_partner'::public.user_role) THEN
        RETURN;
    END IF;

    -- Atualiza o perfil para 'delivery_partner' e 'active'
    UPDATE public.user_profiles
    SET role = 'delivery_partner'::public.user_role,
        status = 'active'::public.user_status
    WHERE id = v_user_id;

    -- Garante a criação do perfil de parceiro, se não existir
    INSERT INTO public.partner_profiles (user_id, verification_status, is_available)
    VALUES (v_user_id, 'PENDING_REVIEW', FALSE)
    ON CONFLICT (user_id) DO NOTHING;

    -- Garante a criação da carteira de parceiro, se não existir
    INSERT INTO public.partner_wallets (partner_id)
    VALUES (v_user_id)
    ON CONFLICT (partner_id) DO NOTHING;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5.3 Ajuste Manual de Saldo (Admin)
CREATE OR REPLACE FUNCTION public.adjust_wallet_balance(p_user_id uuid, p_amount float, p_reason text)
RETURNS void AS $$
DECLARE
    v_role public.user_role;
    v_current_user_role public.user_role;
BEGIN
    -- Obter a role do usuário atualmente autenticado
    SELECT role INTO v_current_user_role FROM public.user_profiles WHERE id = auth.uid();

    -- Verificar se quem chama é admin
    IF v_current_user_role IS NULL OR v_current_user_role != 'admin'::public.user_role THEN
        RAISE EXCEPTION 'Acesso negado: Apenas administradores podem ajustar o saldo.';
    END IF;

    -- Pegar role do usuário alvo
    SELECT role INTO v_role FROM public.user_profiles WHERE id = p_user_id;

    IF v_role = 'store_partner'::public.user_role THEN
        UPDATE public.store_wallets 
        SET balance = balance + p_amount, 
            balance_decimal = balance_decimal + p_amount 
        WHERE store_id = p_user_id;
        
        INSERT INTO public.wallet_transactions (store_id, amount, type, description, status)
        VALUES (p_user_id, p_amount, 'ADJUSTMENT', p_reason, 'COMPLETED');

    ELSIF v_role = 'delivery_partner'::public.user_role THEN
        UPDATE public.partner_wallets 
        SET balance = balance + p_amount 
        WHERE partner_id = p_user_id;

        INSERT INTO public.wallet_transactions (partner_id, amount, type, description, status)
        VALUES (p_user_id, p_amount, 'ADJUSTMENT', p_reason, 'COMPLETED');
    
    ELSE
        RAISE EXCEPTION 'Usuário alvo não possui carteira ativa (não é loja nem parceiro).';
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 6.1 Trigger para novos usuários (Auth -> Profiles/Wallets/Notifications)
CREATE OR REPLACE FUNCTION public.handle_new_user() 
RETURNS trigger AS $$
BEGIN
  -- Insere na user_profiles
  INSERT INTO public.user_profiles (id, email, name, role, phone_number, city)
  VALUES (
      new.id, 
      new.email, 
      new.raw_user_meta_data->>'name', 
      coalesce((new.raw_user_meta_data->>'role')::public.user_role, 'user'::public.user_role),
      new.raw_user_meta_data->>'phone',
      new.raw_user_meta_data->>'city'
  )
  ON CONFLICT (id) DO UPDATE SET -- Para idempotência em caso de re-insertions ou updates
    email = new.email, 
    name = coalesce(new.raw_user_meta_data->>'name', user_profiles.name),
    role = coalesce((new.raw_user_meta_data->>'role')::public.user_role, user_profiles.role),
    phone_number = coalesce(new.raw_user_meta_data->>'phone', user_profiles.phone_number),
    city = coalesce(new.raw_user_meta_data->>'city', user_profiles.city);
  
  -- Cria carteiras e perfil de parceiro/loja com base na role
  IF (new.raw_user_meta_data->>'role')::public.user_role = 'store_partner'::public.user_role THEN
    INSERT INTO public.store_wallets (store_id) VALUES (new.id) ON CONFLICT (store_id) DO NOTHING;
  ELSIF (new.raw_user_meta_data->>'role')::public.user_role = 'delivery_partner'::public.user_role THEN
    INSERT INTO public.partner_wallets (partner_id) VALUES (new.id) ON CONFLICT (partner_id) DO NOTHING;
    INSERT INTO public.partner_profiles (user_id, city, verification_status) 
    VALUES (new.id, new.raw_user_meta_data->>'city', 'NOT_SUBMITTED') -- Status inicial
    ON CONFLICT (user_id) DO NOTHING;
  END IF;
  
  -- Preferências de notificação padrão
  INSERT INTO public.notification_preferences (user_id) 
  VALUES (new.id) 
  ON CONFLICT (user_id) DO NOTHING;

  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
