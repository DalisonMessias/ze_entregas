-- 10. AUTOMATION & TRIGGERS (Criação de Perfil e Carteira)
-- ==================================================================

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  -- Criar Perfil do Usuário
  INSERT INTO public.user_profiles (
    id,
    email,
    name,
    phone_number,
    cpf,
    city,
    role,
    status,
    verification_status
  )
  VALUES (
    new.id,
    new.email,
    new.raw_user_meta_data->>'name',
    new.raw_user_meta_data->>'phone',
    new.raw_user_meta_data->>'cpf',
    new.raw_user_meta_data->>'city',
    COALESCE((new.raw_user_meta_data->>'role')::public.user_role_enum, 'USER'),
    'active',
    CASE
      WHEN (new.raw_user_meta_data->>'role') = 'DELIVERY_PARTNER' THEN 'NOT_SUBMITTED'::public.verification_status_enum
      ELSE 'APPROVED'::public.verification_status_enum
    END
  );

  -- Criar Carteira se necessário
  IF (new.raw_user_meta_data->>'role') = 'STORE_PARTNER' THEN
      INSERT INTO public.store_wallets (store_id) VALUES (new.id);
  ELSIF (new.raw_user_meta_data->>'role') = 'DELIVERY_PARTNER' THEN
      INSERT INTO public.partner_wallets (partner_id) VALUES (new.id);
  END IF;

  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger para executar a função após inserção em auth.users
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();
