-- FUNÇÃO is_admin() - Criada após a tabela user_profiles para evitar erro de dependência.
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS boolean AS $$
  select coalesce(
    (select role = 'ADMIN' from public.user_profiles where id = auth.uid()),
    false
  )
$$ language sql security definer;

-- FUNÇÃO PARA RESOLVER LOGIN (EMAIL, PHONE, CPF)
CREATE OR REPLACE FUNCTION public.resolve_login_email(identifier text)
RETURNS text AS $$
DECLARE
    v_email text;
BEGIN
    -- 1. Check if input is already an email in auth.users
    SELECT email INTO v_email FROM auth.users WHERE email = identifier;
    IF v_email IS NOT NULL THEN
        RETURN v_email;
    END IF;

    -- 2. Check user_profiles for phone or cpf
    SELECT email INTO v_email
    FROM public.user_profiles
    WHERE phone_number = identifier OR cpf = identifier
    LIMIT 1;

    RETURN v_email;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- FUNÇÃO PARA VERIFICAR SE EMAIL EXISTE (Recuperação de Senha)
CREATE OR REPLACE FUNCTION public.check_email_exists(email_input text)
RETURNS boolean AS $$
BEGIN
    RETURN EXISTS (SELECT 1 FROM auth.users WHERE email = email_input);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Função para criar perfil de usuário
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
DECLARE
    v_role public.user_role_enum;
    v_referral_code TEXT;
BEGIN
    v_role := COALESCE((new.raw_user_meta_data->>'role')::public.user_role_enum, 'USER');
    v_referral_code := substr(md5(random()::text), 0, 9);
    
    INSERT INTO public.user_profiles (id, email, name, phone_number, cpf, role, city, referred_by_code, referral_code)
    VALUES (
        new.id,
        new.email,
        new.raw_user_meta_data->>'name',
        new.raw_user_meta_data->>'phone',
        new.raw_user_meta_data->>'cpf',
        v_role,
        new.raw_user_meta_data->>'city',
        new.raw_user_meta_data->>'referred_by_code',
        v_referral_code
    );
    -- Wallet creation logic for store/partner if needed
    IF v_role = 'STORE_PARTNER' THEN
        INSERT INTO public.store_wallets (store_id) VALUES (new.id);
    ELSIF v_role = 'DELIVERY_PARTNER' THEN
        INSERT INTO public.partner_wallets (partner_id) VALUES (new.id);
    END IF;
    RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger para criar perfil
CREATE TRIGGER on_auth_user_created
AFTER INSERT ON auth.users
FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- FUNÇÃO PARA RELATÓRIOS DA LOJA
CREATE OR REPLACE FUNCTION public.get_store_reports_data()
RETURNS json AS $$
DECLARE
    current_store_id UUID := auth.uid();
    report_data JSON;
BEGIN
    SELECT json_build_object(
        'totalRequests', (SELECT count(*) FROM public.partner_requests WHERE store_id = current_store_id),
        'totalValue', (SELECT coalesce(sum(total_charged_store), 0) FROM public.partner_requests WHERE store_id = current_store_id),
        'peakHours', (
            SELECT coalesce(json_agg(h), '[]')
            FROM (
                SELECT 
                    extract(hour from created_at at time zone 'UTC' at time zone 'America/Sao_Paulo')::int as hour,
                    count(*)::int as count
                FROM public.partner_requests
                WHERE store_id = current_store_id
                GROUP BY hour
                ORDER BY hour
            ) h
        ),
        'driverPerformance', (
            SELECT coalesce(json_agg(d), '[]')
            FROM (
                SELECT 
                    pr.partner_id,
                    up.name as partner_name,
                    count(*)::int as count
                FROM public.partner_requests pr
                JOIN public.user_profiles up ON pr.partner_id = up.id
                WHERE pr.store_id = current_store_id AND pr.partner_id IS NOT NULL
                GROUP BY pr.partner_id, up.name
                ORDER BY count DESC
                LIMIT 5
            ) d
        )
    ) INTO report_data;

    RETURN report_data;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- FUNÇÃO PARA ATIVAÇÃO DE SUPER LOJA
CREATE OR REPLACE FUNCTION public.subscribe_to_super_store(p_fee numeric)
RETURNS void AS $$
DECLARE
    v_balance numeric;
BEGIN
    -- Verificar saldo
    SELECT balance_decimal INTO v_balance FROM public.store_wallets WHERE store_id = auth.uid();
    
    IF v_balance < p_fee THEN
        RAISE EXCEPTION 'Saldo insuficiente.';
    END IF;

    -- Debitar
    UPDATE public.store_wallets SET balance_decimal = balance_decimal - p_fee WHERE store_id = auth.uid();

    -- Registrar transação
    INSERT INTO public.store_wallet_transactions 
    (store_id, type, amount, balance_before, balance_after, status, asaas_charge_id)
    VALUES 
    (auth.uid(), 'FEE', p_fee, v_balance, v_balance - p_fee, 'CONFIRMED', 'INTERNAL_SUPER_STORE_FEE');

    -- Atualizar Perfil
    UPDATE public.user_profiles SET is_super_store = TRUE WHERE id = auth.uid();
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- FUNÇÃO PARA LISTAR TODAS AS CARTEIRAS (ADMIN) (NEW)
CREATE OR REPLACE FUNCTION public.admin_get_consolidated_wallets()
RETURNS TABLE (
    user_id UUID,
    name TEXT,
    email TEXT,
    role TEXT,
    balance NUMERIC
) AS $$
BEGIN
    IF NOT public.is_admin() THEN
        RAISE EXCEPTION 'Acesso negado.';
    END IF;

    RETURN QUERY
    SELECT 
        up.id,
        up.name,
        up.email,
        up.role::TEXT,
        COALESCE(sw.balance_decimal, pw.balance, 0)
    FROM public.user_profiles up
    LEFT JOIN public.store_wallets sw ON up.id = sw.store_id
    LEFT JOIN public.partner_wallets pw ON up.id = pw.partner_id
    WHERE up.role IN ('STORE_PARTNER', 'DELIVERY_PARTNER');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- FUNÇÃO PARA AJUSTE MANUAL DE SALDO (ADMIN) (NEW)
CREATE OR REPLACE FUNCTION public.admin_adjust_balance(p_user_id uuid, p_amount numeric, p_reason text)
RETURNS void AS $$
DECLARE
    v_role public.user_role_enum;
    v_current_balance numeric;
    v_new_balance numeric;
BEGIN
    IF NOT public.is_admin() THEN
        RAISE EXCEPTION 'Acesso negado.';
    END IF;

    SELECT role INTO v_role FROM public.user_profiles WHERE id = p_user_id;

    IF v_role = 'STORE_PARTNER' THEN
        SELECT balance_decimal INTO v_current_balance FROM public.store_wallets WHERE store_id = p_user_id;
        v_new_balance := v_current_balance + p_amount;
        
        -- Validar saldo negativo (opcional, pode remover se permitir dívida)
        IF v_new_balance < 0 THEN
             RAISE EXCEPTION 'Saldo insuficiente para a operação.';
        END IF;

        UPDATE public.store_wallets SET balance_decimal = v_new_balance WHERE store_id = p_user_id;
        
        -- Log transaction for store history too
        INSERT INTO public.store_wallet_transactions (store_id, type, amount, balance_before, balance_after, status, asaas_charge_id, metadata)
        VALUES (p_user_id, CASE WHEN p_amount > 0 THEN 'RELOAD' ELSE 'DEBIT' END, p_amount, v_current_balance, v_new_balance, 'CONFIRMED', 'ADMIN_ADJUST', jsonb_build_object('reason', p_reason));

    ELSIF v_role = 'DELIVERY_PARTNER' THEN
        SELECT balance INTO v_current_balance FROM public.partner_wallets WHERE partner_id = p_user_id;
        v_new_balance := v_current_balance + p_amount;

        IF v_new_balance < 0 THEN
             RAISE EXCEPTION 'Saldo insuficiente para a operação.';
        END IF;

        UPDATE public.partner_wallets SET balance = v_new_balance WHERE partner_id = p_user_id;
        
        -- Log transaction for partner history too
        INSERT INTO public.partner_payments (partner_id, amount, status, is_emergency, fail_reason)
        VALUES (p_user_id, ABS(p_amount), 'DONE', FALSE, 'AJUSTE ADMIN: ' || p_reason);

    ELSE
        RAISE EXCEPTION 'Tipo de usuário inválido para carteira.';
    END IF;

    -- Log Geral de Admin
    INSERT INTO public.admin_wallet_logs (admin_id, target_user_id, amount, previous_balance, new_balance, reason)
    VALUES (auth.uid(), p_user_id, p_amount, v_current_balance, v_new_balance, p_reason);

END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
